import { mastra } from "./mastra/index.js";
import { createTutor } from "./mastra/agents.js";
import { studyInput } from "./mastra/workflows.js";
import { assertActive, generateAnswer, modelId, type AgentModel, type Emit, type LabResult } from "./core.js";
export const inputSchema = studyInput;
export async function executeLab(raw: unknown, emit: Emit, signal: AbortSignal): Promise<LabResult> {
  const input = inputSchema.parse(raw); assertActive(signal);
  emit({ type: "stage", text: "Creating and starting the registered two-step study-plan workflow." });
  const run = await mastra.getWorkflow("studyPlan").createRun();
  const response = await run.start({ inputData: input }); assertActive(signal);
  const businessReady = response.status === "success" && response.result.outcome === "ready";
  return { status: businessReady ? "pass" : "needs-attention", summary: businessReady ? "The workflow completed and found a usable study plan." : "Inspect SDK execution separately from the missing-record business outcome.",
    data: { sdkStatus: response.status, runId: run.runId, result: response.status === "success" ? response.result : null, stepIds: Object.keys(response.steps), memoryConfigured: false } };
}
export const liveModel = (_?: unknown) => modelId();
export async function executeLive(raw: unknown, question: string, emit: Emit, signal: AbortSignal, model?: AgentModel) {
  const report = await executeLab(raw, emit, signal);
  return generateAnswer(model ? createTutor(model) : mastra.getAgent("tutor"), `${question}\nWorkflow evidence: ${JSON.stringify(report.data)}. recentLesson is request context, not stored memory.`, emit, signal);
}
export async function dispose() {}

import { createStep, createWorkflow } from "@mastra/core/workflows";
import { z } from "zod";
import { lookupLesson } from "./tools.js";
export const studyInput = z.object({ lessonId: z.number().int().min(1).max(999), minutes: z.number().int().min(10).max(120), recentLesson: z.number().int().min(1).max(999).optional() }).strict();
const contextSchema = studyInput.extend({ record: z.object({ id: z.number(), title: z.string(), prerequisites: z.array(z.number()) }).nullable() });
const lookup = createStep({ id: "lookup", inputSchema: studyInput, outputSchema: contextSchema,
  execute: async ({ inputData }) => {
    const result = await lookupLesson.execute!({ lessonId: inputData.lessonId }, {});
    if (!result || "error" in result) throw new Error("Catalog tool validation failed.");
    return { ...inputData, record: result.record };
  } });
const planSchema = z.object({ outcome: z.enum(["ready", "needs-clarification"]), title: z.string(), actions: z.array(z.string()), minutes: z.number(), suppliedRecentLesson: z.number().nullable() });
const plan = createStep({ id: "plan", inputSchema: contextSchema, outputSchema: planSchema,
  execute: async ({ inputData }) => ({ outcome: inputData.record ? "ready" as const : "needs-clarification" as const,
    title: inputData.record?.title || "Choose an existing lesson", minutes: inputData.minutes, suppliedRecentLesson: inputData.recentLesson || null,
    actions: inputData.record ? [`Review prerequisites: ${inputData.record.prerequisites.join(", ") || "none"}`, `Study ${inputData.record.title} for ${inputData.minutes - 5} minutes`, "Spend 5 minutes explaining the result"] : ["Select a lesson ID from 1 to 10; do not invent a catalog entry."] }) });
export const studyPlan = createWorkflow({ id: "study-plan", inputSchema: studyInput, outputSchema: planSchema }).then(lookup).then(plan).commit();

import { createTool } from "@mastra/core/tools";
import { z } from "zod";
import { course } from "../course.js";
export const lookupLesson = createTool({
  id: "lookup-lesson", description: "Read one public course record by lesson ID; report missing records explicitly.",
  inputSchema: z.object({ lessonId: z.number().int().min(1).max(999) }).strict(),
  outputSchema: z.object({ found: z.boolean(), record: z.object({ id: z.number(), title: z.string(), prerequisites: z.array(z.number()) }).nullable() }),
  execute: async ({ lessonId }) => ({ found: course.some(item => item.id === lessonId), record: course.find(item => item.id === lessonId) || null }),
});

export async function generateAnswer(agent: Agent, question: string, emit: Emit, signal: AbortSignal,
  options: { maxOutputTokens?: number; temperature?: number } = {}) {
  assertActive(signal);
  emit({ type: "stage", text: "Calling the registered agent with generate(); the provider performs the model work." });
  const response = await agent.generate(question, { maxSteps: MAX_STEPS, abortSignal: signal,
    modelSettings: { maxOutputTokens: options.maxOutputTokens || 1200, ...(options.temperature === undefined ? {} : { temperature: options.temperature }) } });
  assertActive(signal);
  if (response.error) throw response.error;
  if (response.text) emit({ type: "answer", text: response.text });
  return { finished: response.finishReason === "stop" && response.text.trim().length > 0,
    text: response.text, steps: response.steps.length, finishReason: response.finishReason,
    toolCalls: response.toolCalls.length, details: { finishReason: response.finishReason, toolCalls: response.toolCalls.length } };
}
export function resultMarkdown(result: LabResult) {
  return `## ${result.status === "pass" ? "Practical result" : "Review the result"}\n\n${result.summary}\n\n\`\`\`json\n${JSON.stringify(result.data, null, 2)}\n\`\`\``;
}
